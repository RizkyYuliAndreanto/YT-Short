from dataclasses import dataclass
from datetime import datetime
from enum import Enum
from typing import Optional
from decimal import Decimal
from uuid import UUID, uuid4

class OrderStatus(Enum):
    """Valid order states - enforces state machine"""
    PENDING = "PENDING"           # Initial state after creation
    CONFIRMED = "CONFIRMED"       # Customer confirmed
    PROCESSING = "PROCESSING"     # Being prepared/shipped
    COMPLETED = "COMPLETED"       # Final successful state
    CANCELLED = "CANCELLED"       # Final cancelled state
    FAILED = "FAILED"             # Processing failed, can retry

@dataclass(frozen=True)
class Order:
    """
    Domain Entity: Order
    
    Invariants (MUST ALWAYS HOLD):
    - order_id is unique UUID, immutable
    - amount > 0 and Decimal(2 places)
    - customer_id is valid UUID
    - created_at <= updated_at (temporal consistency)
    - status must be valid OrderStatus enum
    - Can only exist in valid state (no intermediate states)
    
    State Machine:
    PENDING → CONFIRMED → PROCESSING → COMPLETED
    PENDING → CANCELLED (at any time before processing)
    PROCESSING → FAILED (if processing fails)
    FAILED → PENDING (can retry)
    """
    
    order_id: UUID
    customer_id: UUID
    amount: Decimal
    status: OrderStatus
    created_at: datetime
    updated_at: datetime
    items: list['OrderItem']
    
    # Derived property (not persisted)
    @property
    def total_items(self) -> int:
        """Derived from items list"""
        return len(self.items)
    
    def __post_init__(self):
        """Validate invariants at construction time"""
        # Amount validation
        if self.amount <= 0:
            raise ValueError(f"Amount must be > 0, got {self.amount}")
        
        if self.amount.as_tuple().exponent < -2:
            raise ValueError(f"Amount can't have more than 2 decimals")
        
        # Temporal validation
        if self.created_at > self.updated_at:
            raise ValueError("created_at must be <= updated_at")
        
        # Status validation
        if self.status not in OrderStatus:
            raise ValueError(f"Invalid status: {self.status}")
        
        # Items validation
        if not self.items:
            raise ValueError("Order must have at least one item")
        
        # Customer validation
        if not self.customer_id:
            raise ValueError("customer_id is required")

@dataclass(frozen=True)
class OrderItem:
    """Immutable order item - frozen dataclass prevents modification"""
    item_id: UUID
    product_id: UUID
    quantity: int
    unit_price: Decimal
    
    def __post_init__(self):
        if self.quantity <= 0:
            raise ValueError(f"Quantity must be > 0, got {self.quantity}")
        if self.unit_price < 0:
            raise ValueError(f"Unit price cannot be negative")

# Factory for creating valid orders
class OrderFactory:
    """Creates Order entities with all invariants satisfied"""
    
    @staticmethod
    def create(
        customer_id: UUID,
        amount: Decimal,
        items: list[OrderItem]
    ) -> Order:
        """
        Factory method ensures valid creation.
        
        Assumptions:
        - customer_id is valid (validated by caller)
        - amount matches sum of items
        - items are valid OrderItems
        
        Failure modes:
        - ValueError if invariants violated
        """
        now = datetime.utcnow()
        
        return Order(
            order_id=uuid4(),
            customer_id=customer_id,
            amount=amount,
            status=OrderStatus.PENDING,
            created_at=now,
            updated_at=now,
            items=items
        )
    
    @staticmethod
    def from_confirmed(order: Order) -> Order:
        """
        Transition order to CONFIRMED state.
        
        Preconditions:
        - order.status must be PENDING
        
        Failure modes:
        - ValueError if invalid transition
        """
        if order.status != OrderStatus.PENDING:
            raise ValueError(
                f"Can only confirm PENDING orders, "
                f"current status: {order.status}"
            )
        
        return Order(
            order_id=order.order_id,
            customer_id=order.customer_id,
            amount=order.amount,
            status=OrderStatus.CONFIRMED,
            created_at=order.created_at,
            updated_at=datetime.utcnow(),
            items=order.items
        )

// ============================================================================
// ENTITY LAYER: Order Domain Model
// ============================================================================

/**
 * Valid order statuses - enforces state machine invariant
 * 
 * Transition rules:
 * PENDING → CONFIRMED → PROCESSING → COMPLETED
 * PENDING → CANCELLED
 * PROCESSING → FAILED → PENDING (retry)
 */
export enum OrderStatus {
  PENDING = "PENDING",
  CONFIRMED = "CONFIRMED",
  PROCESSING = "PROCESSING",
  COMPLETED = "COMPLETED",
  CANCELLED = "CANCELLED",
  FAILED = "FAILED",
}

/**
 * Order Entity - Core domain model
 * 
 * INVARIANTS (MUST ALWAYS HOLD):
 * 1. orderId is unique, immutable UUID
 * 2. amount > 0 with max 2 decimal places
 * 3. customerId is valid, non-null UUID
 * 4. createdAt <= updatedAt (temporal consistency)
 * 5. status is valid OrderStatus enum value
 * 6. items array non-empty
 * 7. State transitions follow state machine
 * 8. Cannot modify after COMPLETED/CANCELLED
 * 
 * FAILURE MODES:
 * - Construction with invalid amount → throw
 * - State transition to invalid state → throw
 * - Modification of immutable fields → throw
 * - Temporal invariant violation → throw
 */
export class Order {
  // Immutable properties (set at creation, never change)
  readonly orderId: string;
  readonly customerId: string;
  readonly createdAt: Date;
  
  // Mutable properties (but with rules)
  private _amount: Decimal;
  private _status: OrderStatus;
  private _updatedAt: Date;
  private _items: OrderItem[];
  
  // INVARIANT: Constructor validates all constraints
  constructor(
    orderId: string,
    customerId: string,
    amount: Decimal,
    status: OrderStatus,
    createdAt: Date,
    updatedAt: Date,
    items: OrderItem[]
  ) {
    // INVARIANT VALIDATION: Amount
    if (amount.lessThanOrEqualTo(0)) {
      throw new DomainError(
        "INVALID_AMOUNT",
        `Amount must be > 0, got ${amount}`
      );
    }
    
    // INVARIANT VALIDATION: Decimal precision
    if (amount.decimalPlaces() > 2) {
      throw new DomainError(
        "INVALID_AMOUNT_PRECISION",
        `Amount can have max 2 decimal places`
      );
    }
    
    // INVARIANT VALIDATION: Temporal consistency
    if (createdAt.getTime() > updatedAt.getTime()) {
      throw new DomainError(
        "TEMPORAL_INVARIANT_VIOLATED",
        "createdAt must be <= updatedAt"
      );
    }
    
    // INVARIANT VALIDATION: Status
    if (!Object.values(OrderStatus).includes(status)) {
      throw new DomainError(
        "INVALID_STATUS",
        `Status must be valid OrderStatus, got ${status}`
      );
    }
    
    // INVARIANT VALIDATION: Items
    if (!items || items.length === 0) {
      throw new DomainError(
        "NO_ITEMS",
        "Order must have at least one item"
      );
    }
    
    // INVARIANT VALIDATION: Customer ID
    if (!customerId || !isValidUUID(customerId)) {
      throw new DomainError(
        "INVALID_CUSTOMER_ID",
        `Customer ID must be valid UUID, got ${customerId}`
      );
    }
    
    // All invariants satisfied, assign
    this.orderId = orderId;
    this.customerId = customerId;
    this.createdAt = createdAt;
    this._amount = amount;
    this._status = status;
    this._updatedAt = updatedAt;
    this._items = items;
  }
  
  // Read-only accessors for mutable properties
  get amount(): Decimal {
    return this._amount;
  }
  
  get status(): OrderStatus {
    return this._status;
  }
  
  get updatedAt(): Date {
    return this._updatedAt;
  }
  
  get items(): ReadonlyArray<OrderItem> {
    return Object.freeze([...this._items]);
  }
  
  /**
   * Transition to CONFIRMED status
   * 
   * PRECONDITION: status must be PENDING
   * POSTCONDITION: new Order with status=CONFIRMED returned
   * 
   * FAILURE MODES:
   * - DomainError if invalid transition
   * 
   * NOTE: Returns NEW Order (immutable pattern), doesn't mutate
   */
  confirmOrder(): Order {
    if (this._status !== OrderStatus.PENDING) {
      throw new DomainError(
        "INVALID_STATE_TRANSITION",
        `Can only confirm PENDING orders, current status: ${this._status}`
      );
    }
    
    return new Order(
      this.orderId,
      this.customerId,
      this._amount,
      OrderStatus.CONFIRMED,
      this.createdAt,
      new Date(),
      this._items
    );
  }
  
  /**
   * Transition to CANCELLED status
   * 
   * PRECONDITION: status must be PENDING or CONFIRMED
   * POSTCONDITION: new Order with status=CANCELLED returned
   */
  cancelOrder(): Order {
    const cancellableStates = [OrderStatus.PENDING, OrderStatus.CONFIRMED];
    
    if (!cancellableStates.includes(this._status)) {
      throw new DomainError(
        "CANNOT_CANCEL",
        `Cannot cancel order in status: ${this._status}`
      );
    }
    
    return new Order(
      this.orderId,
      this.customerId,
      this._amount,
      OrderStatus.CANCELLED,
      this.createdAt,
      new Date(),
      this._items
    );
  }
  
  /**
   * Check if order can be modified
   * Orders in terminal states cannot be modified
   */
  canModify(): boolean {
    const terminalStates = [OrderStatus.COMPLETED, OrderStatus.CANCELLED];
    return !terminalStates.includes(this._status);
  }
}

export class OrderItem {
  constructor(
    readonly itemId: string,
    readonly productId: string,
    readonly quantity: number,
    readonly unitPrice: Decimal
  ) {
    if (quantity <= 0) {
      throw new DomainError(
        "INVALID_QUANTITY",
        `Quantity must be > 0, got ${quantity}`
      );
    }
    
    if (unitPrice.isNegative()) {
      throw new DomainError(
        "INVALID_UNIT_PRICE",
        `Unit price cannot be negative`
      );
    }
  }
}

   * Create a new Order entity

   * FAILURE MODES: DomainError if input violates invariants
   */
  static createOrder(

    return new Order(

      now,

}

  ) {

  }

from abc import ABC, abstractmethod
from typing import Optional, List
from datetime import datetime
from uuid import UUID
import asyncio
from sqlalchemy import select, and_
from sqlalchemy.ext.asyncio import AsyncSession

class OrderRepository(ABC):
    """
    Repository abstraction for Order persistence.
    
    Contract: All methods must handle database failures gracefully.
    Failure modes handled:
    - ConnectionError: connection lost
    - TimeoutError: operation exceeds timeout
    - IntegrityError: constraint violation
    """
    
    @abstractmethod
    async def save(self, order: Order) -> Order:
        """
        Persist order to database.
        
        Preconditions:
        - order must satisfy all invariants
        
        Postconditions:
        - order is persisted with assigned ID
        
        Failure modes:
        - DuplicateKeyError: order_id already exists
        - ConnectionError: database unavailable
        - TimeoutError: operation exceeds timeout
        """
        pass
    
    @abstractmethod
    async def find_by_id(self, order_id: UUID) -> Optional[Order]:
        """
        Retrieve order by ID.
        
        Performance: O(1) with primary key index
        
        Failure modes:
        - ConnectionError: database unavailable
        - TimeoutError: query exceeds timeout
        """
        pass
    
    @abstractmethod
    async def find_by_customer_id(
        self,
        customer_id: UUID,
        limit: int = 100,
        offset: int = 0
    ) -> List[Order]:
        """
        Retrieve orders for customer (paginated).
        
        Performance: O(n) with index on customer_id
        
        Failure modes:
        - ConnectionError: database unavailable
        - TimeoutError: query exceeds timeout
        """
        pass
    
    @abstractmethod
    async def update(self, order: Order) -> Order:
        """
        Update existing order.
        
        Preconditions:
        - order must exist in database
        - order must satisfy all invariants
        
        Failure modes:
        - NotFoundError: order doesn't exist
        - ConcurrentModificationError: order modified since read
        - ConnectionError: database unavailable
        """
        pass


class SQLAlchemyOrderRepository(OrderRepository):
    """
    SQL implementation of OrderRepository.
    
    Technology: SQLAlchemy with async session
    Database: PostgreSQL (with connection pooling)
    Timeout: 5 seconds per operation
    """
    
    def __init__(self, session: AsyncSession):
        self.session = session
        self.timeout_seconds = 5
    
    async def save(self, order: Order) -> Order:
        """
        Persist new order.
        
        Error handling:
        1. IntegrityError (duplicate key) → convert to DomainError
        2. ConnectionError → retry with backoff
        3. TimeoutError → fail immediately
        """
        try:
            # Create database model from entity
            db_order = OrderModel(
                order_id=order.order_id,
                customer_id=order.customer_id,
                amount=order.amount,
                status=order.status.value,
                created_at=order.created_at,
                updated_at=order.updated_at
            )
            
            # Add child items
            for item in order.items:
                db_item = OrderItemModel(
                    item_id=item.item_id,
                    product_id=item.product_id,
                    quantity=item.quantity,
                    unit_price=item.unit_price
                )
                db_order.items.append(db_item)
            
            self.session.add(db_order)
            
            # Flush to detect constraint violations
            await asyncio.wait_for(
                self.session.flush(),
                timeout=self.timeout_seconds
            )
            
            await asyncio.wait_for(
                self.session.commit(),
                timeout=self.timeout_seconds
            )
            
            # Return persisted entity
            return order  # ID already assigned by entity
            
        except sqlalchemy.exc.IntegrityError as e:
            await self.session.rollback()
            if "order_id" in str(e):
                raise RepositoryError(
                    "DUPLICATE_ORDER_ID",
                    f"Order {order.order_id} already exists"
                )
            raise RepositoryError(
                "CONSTRAINT_VIOLATION",
                f"Constraint violation: {str(e)}"
            )
        
        except asyncio.TimeoutError:
            await self.session.rollback()
            raise RepositoryError(
                "TIMEOUT",
                f"Save operation exceeded {self.timeout_seconds}s timeout"
            )
        
        except Exception as e:
            await self.session.rollback()
            raise RepositoryError(
                "UNKNOWN",
                f"Unexpected error during save: {str(e)}"
            )
    
    async def find_by_id(self, order_id: UUID) -> Optional[Order]:
        """Query with timeout and error handling"""
        try:
            query = select(OrderModel).where(
                OrderModel.order_id == order_id
            )
            
            result = await asyncio.wait_for(
                self.session.execute(query),
                timeout=self.timeout_seconds
            )
            
            db_order = result.scalar_one_or_none()
            
            if not db_order:
                return None
            
            # Convert database model back to entity
            return db_order.to_entity()
            
        except asyncio.TimeoutError:
            raise RepositoryError(
                "TIMEOUT",
                f"Query exceeded {self.timeout_seconds}s timeout"
            )
        
        except Exception as e:
            raise RepositoryError(
                "QUERY_ERROR",
                f"Error querying order: {str(e)}"
            )
    
    async def find_by_customer_id(
        self,
        customer_id: UUID,
        limit: int = 100,
        offset: int = 0
    ) -> List[Order]:
        """
        Query orders for customer.
        
        Performance notes:
        - Requires index on (customer_id, created_at DESC)
        - Limit 100 prevents large result sets
        - Pagination prevents memory exhaustion
        """
        try:
            query = select(OrderModel).where(
                OrderModel.customer_id == customer_id
            ).order_by(
                OrderModel.created_at.desc()
            ).limit(limit).offset(offset)
            
            result = await asyncio.wait_for(
                self.session.execute(query),
                timeout=self.timeout_seconds
            )
            
            db_orders = result.scalars().all()
            
            return [order.to_entity() for order in db_orders]
            
        except asyncio.TimeoutError:
            raise RepositoryError(
                "TIMEOUT",
                f"Query exceeded {self.timeout_seconds}s timeout"
            )
        
        except Exception as e:
            raise RepositoryError(
                "QUERY_ERROR",
                f"Error querying orders: {str(e)}"
            )
    
    async def update(self, order: Order) -> Order:
        """
        Update existing order.
        
        Error handling:
        1. Not found → DomainError
        2. Concurrent modification → OptimisticLockError
        3. Constraint violation → DomainError
        """
        try:
            # Fetch existing to verify exists
            existing = await self.find_by_id(order.order_id)
            
            if not existing:
                raise RepositoryError(
                    "NOT_FOUND",
                    f"Order {order.order_id} not found"
                )
            
            # Update fields
            db_order = await self.session.get(
                OrderModel,
                order.order_id
            )
            
            db_order.status = order.status.value
            db_order.updated_at = order.updated_at
            
            await asyncio.wait_for(
                self.session.commit(),
                timeout=self.timeout_seconds
            )
            
            return order
            
        except asyncio.TimeoutError:
            await self.session.rollback()
            raise RepositoryError(
                "TIMEOUT",
                f"Update exceeded {self.timeout_seconds}s timeout"
            )
        
        except Exception as e:
            await self.session.rollback()
            raise RepositoryError(
                "UPDATE_ERROR",
                f"Error updating order: {str(e)}"
            )


class RepositoryError(Exception):
    """Error from data access layer"""
    def __init__(self, code: str, message: str):
        self.code = code
        super().__init__(message)

from dataclasses import dataclass
from typing import Optional
from decimal import Decimal
from uuid import UUID
import logging
from enum import Enum
from datetime import datetime

logger = logging.getLogger(__name__)

class ConfirmOrderCommand:
    """
    COMMAND: Confirm an order
    
    Business Operation:
    Customer confirms their pending order, transitioning it to CONFIRMED state.
    This locks the order in and prevents cancellation in most cases.
    
    Preconditions:
    - Order must exist
    - Order must be in PENDING status
    - Customer must be verified
    - Inventory must be available
    
    Postconditions (Success):
    - Order status changed to CONFIRMED
    - Order.updated_at set to current time
    - Inventory reserved
    - Order persisted
    
    Failure Modes:
    1. Order not found → OrderNotFoundError (permanent)
    2. Order already confirmed → OrderAlreadyConfirmedError (permanent)
    3. Customer limit exceeded → CustomerLimitExceededError (permanent)
    4. Inventory unavailable → InventoryUnavailableError (transient)
    5. Database timeout → RepositoryTimeoutError (transient)
    6. External service timeout → ExternalServiceError (transient)
    
    Idempotency:
    - Calling twice with same order_id:
      First call: order confirmed
      Second call: order already confirmed, return success (idempotent)
    """
    
    def __init__(
        self,
        order_repository: OrderRepository,
        inventory_service: 'InventoryService',
        customer_service: 'CustomerService',
        logger: logging.Logger
    ):
        """
        Dependency injection.
        
        All dependencies required for business logic.
        """
        self.order_repository = order_repository
        self.inventory_service = inventory_service
        self.customer_service = customer_service
        self.logger = logger
    
    async def execute(
        self,
        order_id: UUID,
        customer_id: UUID
    ) -> ConfirmOrderResult:
        """
        Execute confirmation logic.
        
        INPUT:
        - order_id: UUID of order to confirm
        - customer_id: UUID of customer owning order
        
        OUTPUT:
        - ConfirmOrderResult with status and order data
        
        EXCEPTIONS:
        - Raises DomainError for permanent failures
        - Raises ServiceError for transient failures
        """
        
        self.logger.info(
            "Confirming order",
            extra={"order_id": order_id, "customer_id": customer_id}
        )
        
        try:
            # Step 1: Fetch order
            order = await self._fetch_order(order_id)
            
            # Step 2: Verify ownership
            if order.customer_id != customer_id:
                self.logger.warning(
                    "Customer attempting to confirm order they don't own",
                    extra={"order_id": order_id, "customer_id": customer_id}
                )
                raise DomainError(
                    "UNAUTHORIZED",
                    "Customer does not own this order"
                )
            
            # Step 3: Check order state
            if order.status != OrderStatus.PENDING:
                self.logger.info(
                    "Order not in PENDING status",
                    extra={
                        "order_id": order_id,
                        "current_status": order.status
                    }
                )
                # Idempotent: if already confirmed, return success
                if order.status == OrderStatus.CONFIRMED:
                    return ConfirmOrderResult(
                        status="ALREADY_CONFIRMED",
                        order=order
                    )
                else:
                    raise DomainError(
                        "INVALID_STATE",
                        f"Cannot confirm order in {order.status} state"
                    )
            
            # Step 4: Verify customer creditworthiness
            customer_ok = await self._verify_customer(customer_id, order.amount)
            if not customer_ok:
                self.logger.warning(
                    "Customer credit limit exceeded",
                    extra={"customer_id": customer_id, "amount": order.amount}
                )
                raise DomainError(
                    "CUSTOMER_LIMIT_EXCEEDED",
                    "Customer has exceeded credit limit"
                )
            
            # Step 5: Reserve inventory
            reserved = await self._reserve_inventory(order)
            if not reserved:
                self.logger.warning(
                    "Inventory unavailable for order",
                    extra={"order_id": order_id}
                )
                raise ServiceError(
                    "INVENTORY_UNAVAILABLE",
                    "Requested items out of stock"
                )
            
            # Step 6: Transition order to CONFIRMED
            confirmed_order = order.confirm_order()
            
            # Step 7: Persist
            saved_order = await self._save_order(confirmed_order)
            
            self.logger.info(
                "Order confirmed successfully",
                extra={"order_id": order_id}
            )
            
            return ConfirmOrderResult(
                status="SUCCESS",
                order=saved_order
            )
        
        except DomainError:
            # Domain errors: permanent, don't retry
            self.logger.error(
                "Domain error confirming order",
                exc_info=True,
                extra={"order_id": order_id}
            )
            raise
        
        except ServiceError:
            # Service errors: transient, can retry
            self.logger.warning(
                "Service error confirming order (retryable)",
                exc_info=True,
                extra={"order_id": order_id}
            )
            raise
        
        except Exception as e:
            # Unexpected error
            self.logger.exception(
                "Unexpected error confirming order",
                extra={"order_id": order_id}
            )
            raise ServiceError(
                "INTERNAL_ERROR",
                "Unexpected error during order confirmation"
            )
    
    # ========== HELPER METHODS (Sub-operations) ==========
    
    async def _fetch_order(self, order_id: UUID) -> Order:
        """
        Fetch order from repository.
        
        Failure modes:
        - RepositoryError (timeout, connection) → ServiceError
        - Order not found → DomainError
        """
        try:
            order = await self.order_repository.find_by_id(order_id)
            
            if not order:
                raise DomainError(
                    "ORDER_NOT_FOUND",
                    f"Order {order_id} not found"
                )
            
            return order
        
        except RepositoryError as e:
            if e.code == "TIMEOUT":
                raise ServiceError(
                    "DATABASE_TIMEOUT",
                    "Database unavailable, please retry"
                )
            raise ServiceError(
                "DATABASE_ERROR",
                "Error accessing database"
            )
    
    async def _verify_customer(
        self,
        customer_id: UUID,
        amount: Decimal
    ) -> bool:
        """
        Verify customer is creditworthy.
        
        Returns: True if customer can afford, False otherwise
        
        Failure modes:
        - ServiceError on customer service timeout/error
        """
        try:
            customer = await self.customer_service.get_customer(customer_id)
            
            if not customer:
                # Unknown customer: reject as safe default
                return False
            
            return customer.available_credit >= amount
        
        except Exception as e:
            self.logger.error(
                "Error verifying customer",
                exc_info=True,
                extra={"customer_id": customer_id}
            )
            raise ServiceError(
                "CUSTOMER_SERVICE_ERROR",
                "Unable to verify customer creditworthiness"
            )
    
    async def _reserve_inventory(self, order: Order) -> bool:
        """
        Reserve inventory for order.
        
        Returns: True if all items reserved, False if any unavailable
        
        Failure modes:
        - ServiceError on inventory service timeout/error
        """
        try:
            items_to_reserve = [
                (item.product_id, item.quantity)
                for item in order.items
            ]
            
            result = await self.inventory_service.reserve_items(
                items_to_reserve
            )
            
            return result.all_reserved
        
        except Exception as e:
            self.logger.error(
                "Error reserving inventory",
                exc_info=True,
                extra={"order_id": order.order_id}
            )
            raise ServiceError(
                "INVENTORY_SERVICE_ERROR",
                "Unable to reserve inventory"
            )
    
    async def _save_order(self, order: Order) -> Order:
        """
        Persist order to repository.
        
        Failure modes:
        - RepositoryError → ServiceError
        """
        try:
            return await self.order_repository.update(order)
        
        except RepositoryError as e:
            if e.code == "TIMEOUT":
                raise ServiceError(
                    "DATABASE_TIMEOUT",
                    "Database timeout, please retry"
                )
            raise ServiceError(
                "DATABASE_ERROR",
                "Error saving order"
            )


# Result object (LAYER 7 partial)
@dataclass
class ConfirmOrderResult:
    """Result of order confirmation"""
    status: str  # "SUCCESS" | "ALREADY_CONFIRMED" | "ERROR"
    order: Optional[Order] = None
    error_code: Optional[str] = None
    error_message: Optional[str] = None


# Error types
class DomainError(Exception):
    """Permanent, non-retryable error"""
    def __init__(self, code: str, message: str):
        self.code = code
        super().__init__(message)

class ServiceError(Exception):
    """Transient, potentially retryable error"""
    def __init__(self, code: str, message: str):
        self.code = code
        super().__init__(message)
